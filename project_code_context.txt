def create_llm(temperature: float = 0.0, timeout: int = 120):
    """Ưu tiên DeepSeek API gốc nếu có DEEPSEEK_API_KEY, ngược lại dùng OpenRouter."""
    from langchain_openai import ChatOpenAI

    if DEEPSEEK_API_KEY:
        return ChatOpenAI(
            model=DEEPSEEK_MODEL,
            api_key=DEEPSEEK_API_KEY,
            base_url=DEEPSEEK_BASE_URL,
            temperature=temperature,
            max_retries=3,
            timeout=timeout,
        )
    if OPENROUTER_API_KEY:
        return ChatOpenAI(
            model=OPENROUTER_CHAT_MODEL,
            api_key=OPENROUTER_API_KEY,
            base_url=OPENROUTER_BASE_URL,
            temperature=temperature,
            max_retries=3,
            timeout=timeout,
        )
    raise RuntimeError(
        "Thiếu khóa LLM. Khai báo DEEPSEEK_API_KEY hoặc OPENROUTER_API_KEY trong backend/.env."
    )

"""Công đoạn 3 - Khối Symbolic: xây dựng Đồ thị Tri thức pháp lý trên Neo4j.

Kiến trúc trích xuất 2 tầng (LLM + kiểm chứng Symbolic):
  Tầng 1 (tất định) : legal_parser bóc tách Văn bản -> Điều -> Khoản -> Điểm, khung phạt tiền
                      (regex), hình thức xử phạt bổ sung / biện pháp khắc phục và tham chiếu chéo.

def _key(*parts: str) -> str:
    raw = "\0".join(" ".join(str(p).casefold().split()) for p in parts)
    return hashlib.sha1(raw.encode("utf-8")).hexdigest()[:16]


def _clean(text: str, limit: int = 600) -> str:
    text = " ".join((text or "").split())
    return text if len(text) <= limit else text[: limit - 1] + "…"


def _parse_json(content: Any) -> dict:
    if isinstance(content, list):
        content = "\n".join(str(i.get("text", "")) if isinstance(i, dict) else str(i) for i in content)
    text = str(content or "").strip()
    fenced = re.search(r"```(?:json)?\s*(.*?)```", text, re.S | re.I)
    if fenced:
        text = fenced.group(1)
    start, end = text.find("{"), text.rfind("}")
    if start == -1 or end == -1:
        raise ValueError("Không tìm thấy JSON object trong phản hồi LLM")
    return json.loads(text[start : end + 1])


def _clause_kind(clause: Clause) -> str:
    head = clause.text[:60].casefold()
    if head.startswith("hình thức xử phạt bổ sung"):
        return "SUPPLEMENTARY"
    if head.startswith("biện pháp khắc phục hậu quả"):

    if clause.fine_min is not None or clause.warning:
        return "PENALTY"
    return "OTHER"


_SANCTION_REF = re.compile(
    r"((?:điểm\s+[a-zđ](?:\s*(?:,|và)\s*(?:điểm\s+)?[a-zđ](?![a-zà-ỹ]))*\s+)?"
    r"(?:các\s+)?khoản\s+\d+(?:\s*(?:,|và)\s*(?:khoản\s+)?\d+)*)\s+(?:của\s+)?Điều này",
    re.IGNORECASE,
)


def _sanction_targets(text: str) -> list[tuple[int, str | None]]:
    """'điểm b khoản 4 Điều này' -> [(4,'b')]; 'các khoản 1 và 2 Điều này' -> [(1,None),(2,None)]."""
    targets: list[tuple[int, str | None]] = []
    for match in _SANCTION_REF.finditer(text):
        fragment = match.group(1)
        clause_part = fragment[fragment.casefold().find("khoản"):]
        clauses = [int(n) for n in re.findall(r"\d+", clause_part)]
        letters = re.findall(r"điểm\s+([a-zđ])|(?:,|và)\s*([a-zđ])(?![a-zà-ỹ])",
                             fragment[: fragment.casefold().find("khoản")], re.I)
        letters = [a or b for a, b in letters]
        if letters and len(clauses) == 1:
            targets.extend((clauses[0], letter.lower()) for letter in letters)
        else:
            targets.extend((number, None) for number in clauses)
    return targets


def _fine_action_name(clause: Clause) -> str:
    match = re.search(r"đối với (?:hành vi |một trong các hành vi sau:?)?(.*)", clause.text, re.I)
    return _clean(match.group(1) if match else clause.text, 300).rstrip(".;:")


def _base_subject(doc_code: str, article_number: int) -> str | None:
    config = DECREE_RULES.get(doc_code)
    if not config:
        return None
    low, high = config["scope"]
    if not low <= article_number <= high:
        return None
    return INDIVIDUAL if article_number in config["individual_base_articles"] else config["base_subject"]

def _write_skeleton(tx, document: LegalDocument, article: Article) -> dict[str, int]:
    tx.run(
        """
        MERGE (d:LegalDoc {code: $code})
        SET d.title = $title, d.short = $short, d.type = $type, d.sources = $sources
        MERGE (a:Article {id: $id})
        SET a.doc_code = $code, a.number = $number, a.title = $atitle, a.chapter = $chapter,
            a.text = $text, a.source = $source, a.page = $page, a.label = $label
        MERGE (a)-[:PART_OF]->(d)
        """,
        code=document.code, title=document.title, short=document.short, type=document.doc_type,
        sources=document.sources, id=article.id, number=article.number, atitle=article.title,
        chapter=article.chapter, text=article.text, source=article.source, page=article.page,
        label=f"Điều {article.number} {document.short}",
    )
    if article.references:
        tx.run(
            """
            MATCH (a:Article {id: $id})
            UNWIND $refs AS ref
            MATCH (b:Article {doc_code: $code, number: ref})
            MERGE (a)-[:REFERENCES]->(b)
            """,
            id=article.id, refs=article.references, code=document.code,
        )

    clauses, actions, sanctions = [], [], []
    base_subject = _base_subject(document.code, article.number)
    for clause in article.clauses:
        cid = clause_id(document.code, article.number, clause.number)
        kind = _clause_kind(clause)
        clauses.append({
            "id": cid, "number": clause.number, "text": clause.text, "kind": kind,
            "fine_min": clause.fine_min, "fine_max": clause.fine_max, "warning": clause.warning,
            "label": f"Khoản {clause.number} Điều {article.number}",
        })
        if kind == "PENALTY":
            penalty_type = "FINE" if clause.fine_min is not None else "WARNING"
            penalty = {
                "id": f"{cid}:{penalty_type}", "type": penalty_type,
                "min": clause.fine_min, "max": clause.fine_max, "warning": clause.warning,
                "base_subject": base_subject,
            }
            items = [(p.letter, p.text) for p in clause.points] or [("_", _fine_action_name(clause))]
            for letter, text in items:
                actions.append({
                    "id": f"{cid}:{letter}", "clause_id": cid,
                    "point": None if letter == "_" else letter,
                    "name": _clean(text, 300).rstrip(".;:"), "text": text, "penalty": penalty,
                })
        elif kind in {"SUPPLEMENTARY", "REMEDIAL"}:
            items = [(p.letter, p.text) for p in clause.points] or [("_", clause.text)]
            for letter, text in items:
                description = re.sub(r"^(Hình thức xử phạt bổ sung|Biện pháp khắc phục hậu quả)\s*:?\s*",
                                     "", text, flags=re.I)
                sanctions.append({
                    "id": f"{cid}:{letter}", "type": kind, "description": _clean(description, 500),
                    "targets": [
                        {"clause_id": clause_id(document.code, article.number, num), "point": pt}
                        for num, pt in _sanction_targets(text)
                    ],
                })

    tx.run(
        """
        MATCH (a:Article {id: $article_id})
        UNWIND $clauses AS c
        MERGE (k:Clause {id: c.id})
        SET k.number = c.number, k.text = c.text, k.kind = c.kind, k.fine_min = c.fine_min,
            k.fine_max = c.fine_max, k.warning = c.warning, k.label = c.label
        MERGE (k)-[:PART_OF]->(a)
        """,
        article_id=article.id, clauses=clauses,
    )
    tx.run(
        """
        UNWIND $actions AS x
        MATCH (k:Clause {id: x.clause_id})
        MERGE (act:Action {id: x.id})
        ON CREATE SET act.name = x.name, act.kind = 'VIOLATION', act.source = 'parser'
        SET act.text = x.text, act.point = x.point, act.article_id = $article_id
        MERGE (act)-[r:DEFINED_IN]->(k) SET r.point = x.point
        MERGE (p:Penalty {id: x.penalty.id})
        SET p.type = x.penalty.type, p.min_amount = x.penalty.min, p.max_amount = x.penalty.max,
            p.warning = x.penalty.warning, p.base_subject = x.penalty.base_subject
        MERGE (act)-[:PENALIZED_BY]->(p)
        WITH p, x WHERE x.penalty.base_subject IS NOT NULL
        MERGE (s:Subject {key: x.penalty.base_subject})
        ON CREATE SET s.name = CASE x.penalty.base_subject WHEN 'ORGANIZATION' THEN 'Tổ chức'
                                                           ELSE 'Cá nhân' END,
                      s.category = x.penalty.base_subject
        MERGE (p)-[:APPLIES_TO]->(s)
        """,
        actions=actions, article_id=article.id,
    )
    tx.run(
        """
        UNWIND $sanctions AS s
        MERGE (n:Sanction {id: s.id})
        SET n.type = s.type, n.description = s.description
        WITH n, s UNWIND s.targets AS t
        MATCH (act:Action)-[:DEFINED_IN]->(:Clause {id: t.clause_id})
        WHERE t.point IS NULL OR act.point = t.point
        MERGE (act)-[:HAS_SANCTION]->(n)
        """,
        sanctions=sanctions,
    )
    return {"clauses": len(clauses), "actions": len(actions), "sanctions": len(sanctions)}


def _write_llm(tx, document: LegalDocument, article: Article, extraction: ArticleExtraction) -> dict:
    parsed = {c.number: c for c in article.clauses}
    stats = {"norms": 0, "fine_checked": 0, "fine_mismatch": 0}
    rows = []
    used_ids: set[str] = set()
    for index, norm in enumerate(extraction.norms):
        clause = parsed.get(norm.clause) if norm.clause else None
        if clause and _clause_kind(clause) in {"SUPPLEMENTARY", "REMEDIAL"}:
            continue
        if clause:
            cid = clause_id(document.code, article.number, clause.number)
            point_letters = {p.letter for p in clause.points}
            letter = norm.point if norm.point in point_letters else "_"
            action_id = f"{cid}:{letter}" if (letter != "_" or not clause.points) else f"{cid}:llm{index}"
            if action_id in used_ids:  # nhiều quy phạm trong cùng một Khoản/Điểm
                action_id = f"{cid}:llm{index}"
        else:
            cid, letter, action_id = None, None, f"{article.id}:llm{index}"
        used_ids.add(action_id)

        # Kiểm chứng Symbolic: so khung phạt LLM với regex (regex là ground truth)
        if clause and clause.fine_min is not None and norm.fine and norm.fine.min is not None:
            stats["fine_checked"] += 1
            if (norm.fine.min, norm.fine.max) != (clause.fine_min, clause.fine_max):
                stats["fine_mismatch"] += 1

        rows.append({
            "id": action_id, "clause_id": cid, "point": letter if letter not in (None, "_") else None,
            "name": _clean(norm.name, 300), "kind": norm.kind,
            "subjects": [
                {"key": f"{s.category}:{_key(s.name)}", "name": _clean(s.name, 150), "category": s.category}
                for s in norm.subjects if s.name.strip()
            ],
            "conditions": [{"key": _key(c), "name": _clean(c, 300)} for c in norm.conditions if c.strip()],
            "fine_min": norm.fine.min if norm.fine else None,
            "fine_max": norm.fine.max if norm.fine else None,
        })
    stats["norms"] = len(rows)

    tx.run(
        """
        UNWIND $rows AS r
        MERGE (act:Action {id: r.id})
        SET act.name = r.name, act.kind = r.kind, act.article_id = $article_id,
            act.source = CASE WHEN act.source = 'parser' THEN 'parser+llm' ELSE coalesce(act.source, 'llm') END,
            act.text = coalesce(act.text, r.name), act.point = coalesce(act.point, r.point),
            act.llm_fine_min = r.fine_min, act.llm_fine_max = r.fine_max
        WITH act, r
        CALL (act, r) {
            MATCH (k:Clause {id: r.clause_id})
            MERGE (act)-[:DEFINED_IN]->(k)
        }
        CALL (act, r) {
            MATCH (a:Article {id: $article_id}) WHERE r.clause_id IS NULL
            MERGE (act)-[:DEFINED_IN]->(a)
        }
        CALL (act, r) {
            UNWIND r.subjects AS s
            MERGE (sub:Subject {key: s.key}) ON CREATE SET sub.name = s.name, sub.category = s.category
            MERGE (act)-[:COMMITTED_BY]->(sub)
        }
        CALL (act, r) {
            UNWIND r.conditions AS c
            MERGE (cond:Condition {key: c.key}) ON CREATE SET cond.name = c.name
            MERGE (act)-[:HAS_CONDITION]->(cond)
        }
        """,
        rows=rows, article_id=article.id,
    )
    tx.run(
        """
        MATCH (a:Article {id: $id})
        SET a.llm_done = true, a.llm_version = $version, a.llm_at = $at,
            a.llm_fine_checked = $checked, a.llm_fine_mismatch = $mismatch
        """,
        id=article.id, version=EXTRACTOR_VERSION, at=datetime.now(timezone.utc).isoformat(),
        checked=stats["fine_checked"], mismatch=stats["fine_mismatch"],
    )
    return stats

def _structured_text(article: Article) -> str:
    if not article.clauses:
        return _clean(article.text, 6000)
    lines = []
    for clause in article.clauses:
        if clause.points:
            intro = clause.text.split(" a) ")[0]
            lines.append(f"Khoản {clause.number}: {_clean(intro, 800)}")
            lines.extend(f"   Điểm {p.letter}) {_clean(p.text, 800)}" for p in clause.points)
        else:
            lines.append(f"Khoản {clause.number}: {_clean(clause.text, 1200)}")
    return "\n".join(lines)


def extract_article(llm, document: LegalDocument, article: Article) -> ArticleExtraction:
    prompt = EXTRACTION_PROMPT.format(
        doc_title=document.title, article_heading=article.heading(),
        chapter=article.chapter or "-", structured_text=_structured_text(article),
    )
    last_error: Exception | None = None
    for attempt in range(3):
        try:
            response = llm.invoke(prompt)
            return ArticleExtraction.model_validate(_parse_json(response.content))
        except (ValueError, ValidationError) as exc:
            last_error = exc  # JSON/schema lỗi -> thử lại
        except Exception as exc:
            last_error = exc
            time.sleep(5 * (attempt + 1))
    raise RuntimeError(f"{type(last_error).__name__}: {last_error}")


def _articles_done(driver) -> set[str]:
    with driver.session(database=NEO4J_DATABASE) as session:
        result = session.run(
            "MATCH (a:Article) WHERE a.llm_version = $v RETURN a.id AS id", v=EXTRACTOR_VERSION
        )
        return {record["id"] for record in result}


def run_llm_enrichment(driver, documents: list[LegalDocument], workers: int, force: bool, limit: int | None):
    done = set() if force else _articles_done(driver)
    jobs = [(d, a) for d in documents for a in d.articles if a.id not in done]
    if limit:
        jobs = jobs[:limit]
    if not jobs:
        print("[LLM] Tất cả Điều đã được trích xuất (dùng --force để chạy lại).")
        return
    print(f"[LLM] Trích xuất {len(jobs)} Điều với {workers} luồng song song...")
    llm = create_llm(temperature=0)
    totals = {"norms": 0, "fine_checked": 0, "fine_mismatch": 0, "failed": 0}

    with ThreadPoolExecutor(max_workers=workers) as pool, driver.session(database=NEO4J_DATABASE) as session:
        futures = {pool.submit(extract_article, llm, d, a): (d, a) for d, a in jobs}
        for index, future in enumerate(as_completed(futures), start=1):
            document, article = futures[future]
            try:
                extraction = future.result()
                stats = session.execute_write(_write_llm, document, article, extraction)
                for key in ("norms", "fine_checked", "fine_mismatch"):
                    totals[key] += stats[key]
                print(f"  [{index}/{len(jobs)}] {article.id}: {stats['norms']} quy phạm"
                      + (f", lệch khung phạt {stats['fine_mismatch']}/{stats['fine_checked']}"
                         if stats["fine_checked"] else ""))
            except Exception as exc:
                totals["failed"] += 1
                print(f"  [{index}/{len(jobs)}] {article.id}: LỖI {exc}")

    checked = totals["fine_checked"] or 1
    print("\n[LLM] Tổng kết trích xuất:")
    print(f"  Quy phạm được MERGE      : {totals['norms']}")
    print(f"  Điều lỗi                 : {totals['failed']}")
    print(f"  Độ khớp khung phạt LLM   : {100 * (1 - totals['fine_mismatch'] / checked):.1f}% "
          f"({totals['fine_checked'] - totals['fine_mismatch']}/{totals['fine_checked']}) - regex được ưu tiên")

def embed_actions(driver, batch_size: int = 50) -> None:
    with driver.session(database=NEO4J_DATABASE) as session:
        rows = session.run(
            """
            MATCH (act:Action) WHERE act.embedding IS NULL
            OPTIONAL MATCH (act)-[:DEFINED_IN]->(:Clause)-[:PART_OF]->(a:Article)
            RETURN act.id AS id, act.name AS name, act.text AS text, a.title AS article
            """
        ).data()
    if not rows:
        print("[EMBED] Mọi hành vi đã có embedding.")
        return
    embeddings = get_embeddings()
    print(f"[EMBED] Nhúng {len(rows)} hành vi bằng Gemini ({embeddings.model})...")
    dimensions = None
    for start in range(0, len(rows), batch_size):
        batch = rows[start : start + batch_size]
        texts = [
            f"{r['name']}. {r['text'] if r['text'] and r['text'] != r['name'] else ''} ({r['article'] or ''})"
            for r in batch
        ]
        for attempt in range(6):
            try:
                vectors = embeddings.embed_documents(texts)
                break
            except Exception as exc:
                message = str(exc).upper()
                if ("429" in message or "RESOURCE_EXHAUSTED" in message) and attempt < 5:
                    if "PERDAY" in message:
                        print("[EMBED] Hết quota Gemini theo ngày - dừng, chạy lại sau sẽ tiếp tục.")
                        return
                    print("[EMBED] Bị giới hạn tốc độ, chờ 60 giây...")
                    time.sleep(60)
                    continue
                raise
        dimensions = len(vectors[0])
        with driver.session(database=NEO4J_DATABASE) as session:
            session.run(
                """
                UNWIND $items AS item
                MATCH (act:Action {id: item.id})
                CALL db.create.setNodeVectorProperty(act, 'embedding', item.vector)
                """,
                items=[{"id": r["id"], "vector": v} for r, v in zip(batch, vectors)],
            ).consume()
        print(f"  {min(start + batch_size, len(rows))}/{len(rows)}")

    with driver.session(database=NEO4J_DATABASE) as session:
        session.run(
            f"""
            CREATE VECTOR INDEX {ACTION_VECTOR_INDEX} IF NOT EXISTS
            FOR (n:Action) ON (n.embedding)
            OPTIONS {{indexConfig: {{`vector.dimensions`: {int(dimensions)}, `vector.similarity_function`: 'cosine'}}}}
            """
        ).consume()
    print(f"[EMBED] Đã tạo Vector Index '{ACTION_VECTOR_INDEX}' ({dimensions} chiều).")


def print_stats(driver) -> None:
    with driver.session(database=NEO4J_DATABASE) as session:
        nodes = session.run(
            "MATCH (n) RETURN labels(n)[0] AS label, count(*) AS n ORDER BY n DESC"
        ).data()
        rels = session.run(
            "MATCH ()-[r]->() RETURN type(r) AS type, count(*) AS n ORDER BY n DESC"
        ).data()
    print("\n[THỐNG KÊ ĐỒ THỊ]")
    print("  Node: " + ", ".join(f"{r['label']}={r['n']}" for r in nodes))
    print("  Cạnh: " + ", ".join(f"{r['type']}={r['n']}" for r in rels))


def main() -> int:
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8", errors="replace", line_buffering=True)

    parser = argparse.ArgumentParser(description="Xây dựng đồ thị tri thức Luật CNTT trên Neo4j")
    parser.add_argument("--reset", action="store_true", help="Xóa toàn bộ đồ thị trước khi nạp")
    parser.add_argument("--no-llm", action="store_true", help="Chỉ dựng khung xương tất định")
    parser.add_argument("--skip-embeddings", action="store_true", help="Không nhúng hành vi")
    parser.add_argument("--force", action="store_true", help="Trích xuất LLM lại cả các Điều đã xong")
    parser.add_argument("--workers", type=int, default=4, help="Số luồng gọi LLM song song")
    parser.add_argument("--limit", type=int, default=None, help="Giới hạn số Điều gửi LLM (thử nghiệm)")
    args = parser.parse_args()

    documents = load_corpus(DATA_DIR)
    driver = create_neo4j_driver()
    try:
        driver.verify_connectivity()
    except Exception as exc:
        print(f"[DỪNG] Không kết nối được Neo4j ({exc}). Chạy: docker compose up -d neo4j")
        return 1

    try:
        if args.reset:
            reset_graph(driver)
        setup_schema(driver)

        totals = {"clauses": 0, "actions": 0, "sanctions": 0}
        with driver.session(database=NEO4J_DATABASE) as session:
            for document in documents:
                for article in document.articles:
                    stats = session.execute_write(_write_skeleton, document, article)
                    for key in totals:
                        totals[key] += stats[key]
        print(f"[SKELETON] {sum(len(d.articles) for d in documents)} Điều, {totals['clauses']} Khoản, "
              f"{totals['actions']} hành vi có chế tài, {totals['sanctions']} biện pháp bổ sung/khắc phục.")

        if not args.no_llm:
            run_llm_enrichment(driver, documents, args.workers, args.force, args.limit)
        if not args.skip_embeddings:
            embed_actions(driver)
        print_stats(driver)
        return 0
    except Exception as exc:
        print(f"[DỪNG] {type(exc).__name__}: {exc}")
        return 1
    finally:
        driver.close()


if __name__ == "__main__":
    raise SystemExit(main())

def build_structured_chunks() -> list[Document]:
    """Cắt theo Điều/Khoản và gắn ngữ cảnh phân cấp (Văn bản > Chương > Điều) vào từng chunk.

    Metadata `article_id` trùng với id của node (:Article) trong Neo4j, nhờ đó kết quả tìm
    kiếm vector có thể "nhảy" sang đồ thị tri thức (cầu nối Neural -> Symbolic).
    """
    fallback_splitter = RecursiveCharacterTextSplitter(chunk_size=1500, chunk_overlap=150)
    chunks: list[Document] = []
    for document in load_corpus(PROJECT_DIR / "data"):
        for article in document.articles:
            header = f"[{document.short}] Chương {article.chapter or '-'} - {article.heading()}"
            metadata = {
                "source": article.source,
                "page": article.page,
                "doc_code": document.code,
                "doc_title": document.short,
                "article_id": article.id,
                "article_number": article.number,
                "article_title": article.title,
            }
            if len(article.text) <= MAX_CHUNK_CHARS or not article.clauses:
                pieces = [article.text] if len(article.text) <= MAX_CHUNK_CHARS else fallback_splitter.split_text(article.text)
                for piece in pieces:
                    chunks.append(Document(page_content=f"{header}\n{piece}", metadata={**metadata, "clauses": ""}))
                continue

            # Điều dài: gom các Khoản liền nhau vào chunk ≤ MAX_CHUNK_CHARS, mỗi chunk đều có header
            group: list[str] = []
            numbers: list[int] = []

            def flush():
                if group:
                    label = f"{numbers[0]}-{numbers[-1]}" if len(numbers) > 1 else str(numbers[0])
                    chunks.append(Document(
                        page_content=f"{header}\n" + "\n".join(group),
                        metadata={**metadata, "clauses": label},
                    ))

            for clause in article.clauses:
                text = f"{clause.number}. {clause.text}"
                if group and sum(len(g) for g in group) + len(text) > MAX_CHUNK_CHARS:
                    flush()
                    group, numbers = [], []
                for piece in ([text] if len(text) <= MAX_CHUNK_CHARS else fallback_splitter.split_text(text)):
                    group.append(piece)
                    numbers.append(clause.number)
            flush()
    return chunks


def main():
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8", errors="replace", line_buffering=True)

    if not os.getenv("GEMINI_API_KEY"):
        raise SystemExit("LỖI: Không tìm thấy GEMINI_API_KEY trong backend/.env!")

    print("📥 BƯỚC 1-2: Đọc PDF và cắt theo cấu trúc Điều/Khoản (Structure-aware Chunking)...")
    chunks = build_structured_chunks()
    print(f"✅ Đã cắt văn bản thành {len(chunks)} đoạn (chunks) có ngữ cảnh phân cấp.\n")

    print("🧠 BƯỚC 3: Khởi tạo mô hình AI nhúng (Embedding) của Google Gemini...")
    embeddings = GoogleGenerativeAIEmbeddings(model=EMBEDDING_MODEL)

    print("💾 BƯỚC 4: Đang tạo embedding trước khi cập nhật PostgreSQL...")
    print("⏳ Quá trình này có thể mất vài phút tùy hạn mức API và tốc độ mạng...")

    text_embeddings = []
    batch_size = 100
    for start in range(0, len(chunks), batch_size):
        batch = chunks[start : start + batch_size]
        for attempt in range(5):
            try:
                vectors = embeddings.embed_documents(
                    [document.page_content for document in batch]
                )
                break
            except Exception as exc:
                message = str(exc).upper()
                rate_limited = "429" in message or "RESOURCE_EXHAUSTED" in message
                daily_limit = "PERDAY" in message or re.search(r"RETRY IN \d+H", message)
                
                if rate_limited and daily_limit:
                    raise SystemExit(
                        "Gemini đã hết quota miễn phí theo ngày. Hãy chạy lại sau khi quota reset "
                        "hoặc dùng project có quota phù hợp. Collection chưa bị thay đổi."
                    ) from exc
                    
                if not rate_limited or attempt == 4:
                    raise
                    
                print("Gemini đang giới hạn tốc độ; chờ 60 giây rồi thử lại batch...")
                time.sleep(60)

        text_embeddings.extend(
            (document.page_content, vector)
            for document, vector in zip(batch, vectors)
        )
        print(f"Đã tạo embedding cho {min(start + len(batch), len(chunks))}/{len(chunks)} đoạn.")

    print("Đang thay collection cũ bằng dữ liệu mới...")
    PGVector.from_embeddings(
        text_embeddings=text_embeddings,
        embedding=embeddings,
        metadatas=[document.metadata for document in chunks],
        collection_name=COLLECTION_NAME,
        connection_string=CONNECTION_STRING,
        pre_delete_collection=True,
    )
    print("🎉 HOÀN TẤT! Toàn bộ văn bản luật đã được nhúng vào Database bằng Gemini.")

if __name__ == "__main__":
    main()

"""Bộ phân tích cấu trúc văn bản pháp luật Việt Nam (tất định, không dùng LLM).

Văn bản  ->  Chương  ->  Điều  ->  Khoản  ->  Điểm

Module này được dùng bởi:
- ingest.py       : cắt chunk theo Điều/Khoản kèm ngữ cảnh phân cấp (khối Neural)
- graph_ingest.py : làm "khung xương" cho đồ thị tri thức và đối chiếu kết quả LLM (khối Symbolic)

DOCUMENT_REGISTRY: list[dict[str, str]] = [
    {
        "pattern": r"15-2020",
        "code": "15/2020/NĐ-CP",

        "bưu chính, viễn thông, tần số vô tuyến điện, công nghệ thông tin và giao dịch điện tử",

    },
    {
        "pattern": r"24-2018",
        "code": "24/2018/QH14",
        "title": "Luật An ninh mạng 2018 (Luật số 24/2018/QH14)",
        "short": "Luật An ninh mạng 2018",
        "type": "LAW",
    },
    {
        "pattern": r"67-2006",
        "code": "67/2006/QH11",
        "title": "Luật Công nghệ thông tin 2006 (Luật số 67/2006/QH11)",
        "short": "Luật Công nghệ thông tin 2006",
        "type": "LAW",
    },
]

POINT_LETTERS = "abcdđeghiklmnopqrstuvxy"  # thứ tự điểm trong văn bản pháp luật Việt Nam

_HEADER_RE = re.compile(r"(?m)^\s*\d*\s*CÔNG BÁO\s*/\s*Số[^\n]*$")
_ARTICLE_RE = re.compile(r"(?m)^[ \t]*Điều[ \t]+(\d+)[ \t]*\.[ \t]*(.*)$")
_CHAPTER_RE = re.compile(r"(?m)^[ \t]*Chương[ \t]+([IVXLC]+)[ \t]*$")
_CLAUSE_RE = re.compile(r"(?m)^[ \t]*(\d{1,2})\.[ \t]+")
_POINT_RE = re.compile(r"(?m)^[ \t]*([a-zđ])\)[ \t]+")
_ARTICLE_REF_RE = re.compile(r"Điều\s+(\d{1,3})(?!\s*\d)")

_FINE_RE = re.compile(
    r"Ph\s*ạt\s+ti\s*ền\s+t\s*ừ\s+([\d.\s]+?)\s*đồng\s+đến\s+([\d.\s]+?)\s*đồng",
    re.IGNORECASE,
)
_WARNING_RE = re.compile(r"Ph\s*ạt\s+c\s*ảnh\s+cáo", re.IGNORECASE)

# Sửa lỗi pypdf chèn khoảng trắng giữa âm tiết: "Ph ạt ti ền t ừ" -> "Phạt tiền từ"
_VOWELS = "aàáảãạăằắẳẵặâầấẩẫậeèéẻẽẹêềếểễệiìíỉĩịoòóỏõọôồốổỗộơờớởỡợuùúủũụưừứửữựyỳýỷỹỵ"
_MARKED_VOWELS = "".join(ch for ch in _VOWELS if ch not in "aeiouy")
_MARKED_VOWELS += _MARKED_VOWELS.upper()
_E_FAMILY = "êềếểễệÊỀẾỂỄỆ"
_CONSONANT_ONLY_SPLIT = re.compile(
    rf"(?<![^\W\d_])([bcdđghklmnpqrstvxBCDĐGHKLMNPQRSTVX]{{1,3}}|[gG]i|[qQ]u) (?=[{_MARKED_VOWELS}])"
)
_VOWEL_E_SPLIT = re.compile(rf"(?<=[iyuoIYUO]) (?=[{_E_FAMILY}])")


def normalize_text(text: str) -> str:
    """Chuẩn hóa Unicode + sửa phần lớn lỗi tách âm tiết do trích xuất PDF."""
    text = unicodedata.normalize("NFC", text or "")
    text = text.replace("\r\n", "\n").replace("\r", "\n").replace("\u00a0", " ")
    text = re.sub(r"[ \t]+", " ", text)
    text = _CONSONANT_ONLY_SPLIT.sub(r"\1", text)
    text = _VOWEL_E_SPLIT.sub("", text)
    text = re.sub(r" +\n", "\n", text)
    return text


def parse_amount(raw: str) -> int | None:
    digits = re.sub(r"[^\d]", "", raw or "")
    return int(digits) if digits else None

def _identify_document(pdf: Path, first_page: str) -> dict[str, str]:
    for entry in DOCUMENT_REGISTRY:
        if re.search(entry["pattern"], pdf.name, re.IGNORECASE):
            return entry
    match = re.search(r"Số\s*:\s*(\d+/\d{4}/[A-ZĐ0-9\-]+)", first_page)
    code = match.group(1) if match else pdf.stem
    return {"code": code, "title": code, "short": code, "type": "OTHER"}


def _read_pages(pdf: Path) -> list[str]:
    reader = PdfReader(str(pdf))
    pages = []
    for page in reader.pages:
        text = normalize_text(page.extract_text() or "")
        pages.append(_HEADER_RE.sub("", text))
    return pages


def _sequential_matches(pattern: re.Pattern, text: str, keys: list[str]):
    """Chỉ nhận các mốc Khoản/Điểm xuất hiện đúng thứ tự (1,2,3... hoặc a,b,c...)."""
    accepted = []
    expected = 0
    for match in pattern.finditer(text):
        if expected < len(keys) and match.group(1) == keys[expected]:
            accepted.append(match)
            expected += 1
    return accepted


def _split_points(clause_text: str) -> list[Point]:
    matches = _sequential_matches(_POINT_RE, clause_text, list(POINT_LETTERS))
    points = []
    for index, match in enumerate(matches):
        end = matches[index + 1].start() if index + 1 < len(matches) else len(clause_text)
        body = " ".join(clause_text[match.end() : end].split())
        points.append(Point(letter=match.group(1), text=body))
    return points


def _split_clauses(body: str) -> list[Clause]:
    matches = _sequential_matches(_CLAUSE_RE, body, [str(i) for i in range(1, 100)])
    clauses = []
    for index, match in enumerate(matches):
        end = matches[index + 1].start() if index + 1 < len(matches) else len(body)
        raw = body[match.end() : end].strip()
        flat = " ".join(raw.split())
        clause = Clause(number=int(match.group(1)), text=flat, points=_split_points(raw))
        fine = _FINE_RE.search(flat[:400])
        if fine:
            clause.fine_min = parse_amount(fine.group(1))
            clause.fine_max = parse_amount(fine.group(2))
        clause.warning = bool(_WARNING_RE.search(flat[:200]))
        clauses.append(clause)
    return clauses


def _parse_articles(meta: dict[str, str], pdf: Path, pages: list[str], last_number: int) -> list[Article]:
    offsets, chunks, cursor = [], [], 0
    for page_text in pages:
        offsets.append(cursor)
        chunks.append(page_text)
        cursor += len(page_text) + 1
    full = "\n".join(chunks)

    def page_of(position: int) -> int:
        page = 0
        for index, start in enumerate(offsets):
            if start <= position:
                page = index
        return page + 1

    chapters = [(m.start(), m.group(1)) for m in _CHAPTER_RE.finditer(full)]
    headers = []
    for match in _ARTICLE_RE.finditer(full):
        number = int(match.group(1))
        if number <= last_number:
            continue  # bỏ trích dẫn "Điều x." trong thân văn bản / tham chiếu ngược
        headers.append(match)
        last_number = number

    articles = []
    for index, match in enumerate(headers):
        start = match.start()
        end = headers[index + 1].start() if index + 1 < len(headers) else len(full)
        # Cắt bỏ tiêu đề Chương tiếp theo nằm trong khoảng của Điều hiện tại
        for chapter_pos, _ in chapters:
            if start < chapter_pos < end:
                end = chapter_pos
                break
        block = full[start:end].strip()
        lines = block.split("\n")
        title = match.group(2).strip()
        body_start = 1
        # Tiêu đề Điều có thể xuống dòng (dòng tiếp theo bắt đầu bằng chữ thường)
        while body_start < len(lines) and lines[body_start].strip()[:1].islower():
            title += " " + lines[body_start].strip()
            body_start += 1
        body = "\n".join(lines[body_start:])
        chapter = ""
        for chapter_pos, chapter_name in chapters:
            if chapter_pos < start:
                chapter = chapter_name
        number = int(match.group(1))
        refs = sorted(
            {int(r) for r in _ARTICLE_REF_RE.findall(body) if int(r) != number}
        )
        articles.append(
            Article(
                doc_code=meta["code"],
                number=number,
                title=" ".join(title.split()),
                chapter=chapter,
                text=" ".join(block.split()),
                source=pdf.name,
                page=page_of(start),
                clauses=_split_clauses(body),
                references=refs,
            )
        )
    return articles


def load_corpus(data_dir: Path) -> list[LegalDocument]:
    """Đọc toàn bộ PDF trong data/, gộp các file thuộc cùng một văn bản (VD: 2 số Công báo)."""
    documents: dict[str, LegalDocument] = {}
    for pdf in sorted(data_dir.glob("*.pdf")):
        pages = _read_pages(pdf)
        if sum(len(p.strip()) for p in pages) < 200:
            print(f"[PARSER] Bỏ qua {pdf.name}: PDF dạng ảnh scan, không có lớp văn bản (cần OCR).")
            continue
        meta = _identify_document(pdf, pages[0] if pages else "")
        document = documents.get(meta["code"])
        if document is None:
            document = LegalDocument(
                code=meta["code"],
                title=meta["title"],
                short=meta.get("short", meta["code"]),
                doc_type=meta["type"],
            )
            documents[meta["code"]] = document
        last = document.articles[-1].number if document.articles else 0
        new_articles = _parse_articles(meta, pdf, pages, last)
        document.sources.append(pdf.name)
        document.articles.extend(new_articles)
        print(f"[PARSER] {pdf.name}: {len(new_articles)} Điều -> {document.code}")
    return list(documents.values())

"""Bộ suy luận luật tất định (Deterministic Legal Rule Engine) - linh hồn của khối Symbolic.

Nguyên tắc: LLM KHÔNG được tự tính toán số tiền. Mọi phép tính mức phạt theo chủ thể,

DECREE_RULES: dict[str, dict] = {

        "scope": (5, 113),  # Chương II (Điều 5) -> Chương VII (Điều 113)
        "base_subject": ORGANIZATION,
        "individual_ratio": 0.5,
        "individual_base_articles": {106},  # Điều 106: vi phạm của người chơi (cá nhân)
        "basis": "Điều 4 khoản 3 Nghị định 15/2020/NĐ-CP",

        "individual_caps": [

# Quy tắc độ tuổi - Luật Xử lý vi phạm hành chính 2012 (sửa đổi 2020).
# LƯU Ý: văn bản này CHƯA có trong thư mục data/, nên đánh dấu external=True để minh bạch.
AGE_RULE_BASIS = "Luật Xử lý vi phạm hành chính 2012 (sửa đổi, bổ sung 2020), Điều 5 và Điều 134"


@dataclass
class RuleStep:
    rule: str
    basis: str
    detail: str
    external: bool = False


@dataclass
class QueryFacts:
    subjects: list[str]
    subject_detected: bool
    age: int | None
    article_refs: list[int]
    doc_hints: list[str]
    invalid_docs: list[str] = field(default_factory=list)

        return asdict(self)


# ---------------------------------------------------------------------------
# Hiểu câu hỏi bằng luật (rule-based NLU) - không tốn lời gọi LLM, kết quả tái lập được
# ---------------------------------------------------------------------------
_ORG_PATTERN = re.compile(
    r"\b(công ty|doanh nghiệp|tổ chức|cơ quan|đơn vị|tập đoàn|pháp nhân|nhà mạng|nhà cung cấp|"
    r"ngân hàng|cửa hàng|chi nhánh|startup|website của công ty|fanpage của công ty)"
    r"(\s+(của\s+)?(tôi|mình|chúng tôi|em|anh|chị|bạn))?\b",
    re.IGNORECASE,
)
_IND_PATTERN = re.compile(
    r"\b(cá nhân|tôi|mình|em|anh|chị|người dùng|nhân viên|học sinh|sinh viên|công dân|"
    r"người chơi|một người|người nào|ai đó|bạn)\b",
    re.IGNORECASE,
)
_AGE_PATTERN = re.compile(r"(\d{1,2})\s*tuổi", re.IGNORECASE)
_MINOR_PATTERN = re.compile(r"\b(trẻ em|vị thành niên|chưa thành niên)\b", re.IGNORECASE)
_ARTICLE_PATTERN = re.compile(r"Điều\s+(\d{1,3})", re.IGNORECASE)
_DOC_CODE_PATTERN = re.compile(r"(\d+/\d{4}/(?:NĐ-CP|ND-CP|QH\d{2}))", re.IGNORECASE)
_DECREE_PATTERN = re.compile(r"(?:ngh[iị]\s*[dđ][iị]nh)\s*(\d+)", re.IGNORECASE)


def analyze_question(question: str) -> QueryFacts:
    text = question or ""
    subjects: list[str] = []
    if _ORG_PATTERN.search(text):
        subjects.append(ORGANIZATION)
    # Bỏ cụm "công ty tôi" trước khi tìm chủ thể cá nhân để tránh nhận nhầm
    residual = _ORG_PATTERN.sub(" ", text)
    if _IND_PATTERN.search(residual) or _AGE_PATTERN.search(text) or _MINOR_PATTERN.search(text):
        subjects.append(INDIVIDUAL)
    detected = bool(subjects)
    if not subjects:
        subjects = [ORGANIZATION, INDIVIDUAL]

    age_match = _AGE_PATTERN.search(text)
    age = int(age_match.group(1)) if age_match else None
    if age is not None and INDIVIDUAL not in subjects:
        subjects.append(INDIVIDUAL)

    doc_hints = []
    invalid_docs = []
    out_of_scope_docs = []

    # Nhận diện mã văn bản đầy đủ (ví dụ: 15/2020/NĐ-CP, 24/2018/QH14)
    for code in _DOC_CODE_PATTERN.findall(text):
        normalized_code = code.upper().replace("ND-CP", "NĐ-CP")
        if normalized_code in {"15/2020/NĐ-CP", "24/2018/QH14", "67/2006/QH11"}:
            doc_hints.append(normalized_code)
        else:
            out_of_scope_docs.append(normalized_code)

    # Nhận diện số hiệu Nghị định (ví dụ: Nghị định 15, Nghị định 10000...)
    for dec_num_str in _DECREE_PATTERN.findall(text):
        num = int(dec_num_str)
        if num == 15:
            if "15/2020/NĐ-CP" not in doc_hints:
                doc_hints.append("15/2020/NĐ-CP")
        elif num > 500:
            # Trong lịch sử lập pháp Việt Nam, số nghị định trong 1 năm không bao giờ vượt quá 500
            invalid_docs.append(f"Nghị định {num}")
        else:
            out_of_scope_docs.append(f"Nghị định {num}")

    return QueryFacts(
        subjects=subjects,
        subject_detected=detected,
        age=age,
        article_refs=sorted({int(n) for n in _ARTICLE_PATTERN.findall(text)}),
        doc_hints=doc_hints,
        invalid_docs=invalid_docs,
        out_of_scope_docs=out_of_scope_docs,
    )

def format_vnd(amount: int | float | None) -> str:
    if amount is None:
        return "?"
    return f"{int(round(amount)):,}".replace(",", ".") + " đồng"


def _individual_cap(config: dict, article_number: int):
    for (low, high), cap, field_name in config.get("individual_caps", []):
        if low <= article_number <= high:
            return cap, field_name
    return None, None


def apply_penalty_rules(
    penalty: dict, doc_code: str, article_number: int, facts: QueryFacts
) -> dict:
    """Tính mức phạt áp dụng cho từng chủ thể trong câu hỏi + dấu vết suy luận."""
    base_min, base_max = penalty.get("min_amount"), penalty.get("max_amount")
    steps: list[RuleStep] = []
    results: list[dict] = []

    if penalty.get("type") != "FINE" or base_min is None or base_max is None:
        return {"results": results, "steps": [asdict(s) for s in steps]}

    config = DECREE_RULES.get(doc_code)
    low, high = config["scope"] if config else (0, -1)
    in_scope = config is not None and low <= article_number <= high

    if in_scope and article_number in config["individual_base_articles"]:
        base_subject = INDIVIDUAL
        steps.append(RuleStep(
            "BASE_SUBJECT", config["basis"],
            f"Điều {article_number} là ngoại lệ của Điều 4 khoản 3: khung "
            f"{format_vnd(base_min)} - {format_vnd(base_max)} áp dụng trực tiếp cho cá nhân.",
        ))
    elif in_scope:
        base_subject = config["base_subject"]
        steps.append(RuleStep(
            "BASE_SUBJECT", config["basis"],
            f"Khung {format_vnd(base_min)} - {format_vnd(base_max)} tại Điều {article_number} "
            "là mức phạt áp dụng đối với TỔ CHỨC.",
        ))
    else:
        base_subject = penalty.get("base_subject") or None
        steps.append(RuleStep(
            "BASE_SUBJECT", f"Văn bản {doc_code}",
            "Không có quy tắc quy đổi chủ thể cho văn bản này; giữ nguyên khung phạt gốc.",
        ))

    for subject in facts.subjects:
        amount_min, amount_max, formula = base_min, base_max, "giữ nguyên khung gốc"
        applicable = True

        if in_scope and base_subject == ORGANIZATION and subject == INDIVIDUAL:
            ratio = config["individual_ratio"]
            amount_min, amount_max = base_min * ratio, base_max * ratio
            formula = "Cá nhân = 1/2 × mức phạt Tổ chức"
            steps.append(RuleStep(
                "INDIVIDUAL_HALF", config["basis"],
                f"Cá nhân vi phạm như tổ chức -> {format_vnd(base_min)} × 1/2 = "
                f"{format_vnd(amount_min)}; {format_vnd(base_max)} × 1/2 = {format_vnd(amount_max)}.",
            ))
        elif in_scope and base_subject == INDIVIDUAL and subject == ORGANIZATION:
            applicable = False
            formula = "Không áp dụng cho tổ chức"
            steps.append(RuleStep(
                "NOT_APPLICABLE", config["basis"],
                f"Điều {article_number} chỉ điều chỉnh hành vi của cá nhân (người chơi).",
            ))

        if applicable and subject == INDIVIDUAL and in_scope:
            cap, field_name = _individual_cap(config, article_number)
            if cap is not None:
                ok = amount_max <= cap
                steps.append(RuleStep(
                    "CAP_CHECK", config["cap_basis"],
                    f"Kiểm tra trần lĩnh vực {field_name}: {format_vnd(amount_max)} "
                    f"{'≤' if ok else '>'} {format_vnd(cap)} -> {'hợp lệ' if ok else 'VƯỢT TRẦN, giới hạn lại'}.",
                ))
                if not ok:
                    amount_max = cap
                    amount_min = min(amount_min, cap)

        if applicable and subject == INDIVIDUAL and facts.age is not None:
            age = facts.age
            if age < 14:
                applicable = False
                formula = "Dưới 14 tuổi: không bị xử phạt vi phạm hành chính"
                steps.append(RuleStep("AGE_UNDER_14", AGE_RULE_BASIS,
                                      f"Người {age} tuổi chưa đủ tuổi chịu xử phạt VPHC.", True))
            elif age < 16:
                applicable = False
                formula = "Từ đủ 14 đến dưới 16 tuổi: phạt cảnh cáo, không phạt tiền"
                steps.append(RuleStep("AGE_14_16", AGE_RULE_BASIS,
                                      f"Người {age} tuổi chỉ bị xử phạt khi vi phạm do cố ý và "
                                      "áp dụng hình thức cảnh cáo, không phạt tiền.", True))
            elif age < 18:
                amount_min, amount_max = amount_min / 2, amount_max / 2
                formula += " -> người 16-18 tuổi: không quá 1/2 mức của người thành niên"
                steps.append(RuleStep("AGE_16_18", AGE_RULE_BASIS,
                                      f"Người {age} tuổi: mức phạt tiền tối đa = 1/2 -> "
                                      f"{format_vnd(amount_min)} - {format_vnd(amount_max)}.", True))

        results.append({
            "subject": subject,
            "subject_label": SUBJECT_LABELS.get(subject, subject),
            "applicable": applicable,
            "min_amount": int(round(amount_min)) if applicable else None,
            "max_amount": int(round(amount_max)) if applicable else None,
            "display": (f"{format_vnd(amount_min)} - {format_vnd(amount_max)}" if applicable else formula),
            "formula": formula,
        })

    return {"base_subject": base_subject, "results": results, "steps": [asdict(s) for s in steps]}

def _sanitize_fulltext_query(text: str) -> str:
    """Lọc các ký tự đặc biệt của cú pháp Lucene và loại bỏ stopwords pháp lý để tìm hành vi thực tế."""
    cleaned = re.sub(r'[^\w\s\dà-ỹÀ-Ỹ]', ' ', text)
    words = [w.strip() for w in cleaned.split() if len(w.strip()) > 1]
    
    LEGAL_STOP_WORDS = {
        "là", "gì", "thế", "nào", "cho", "tôi", "hỏi", "bị", "phạt", "bao", "nhiêu", "tiền",
        "nghị", "định", "luật", "thông", "tư", "điều", "khoản", "điểm", "bộ", "số", "của", "và", "theo",
        "về", "có", "không", "như", "được", "tại", "trong", "quy", "với", "hành", "vi", "mức"
    }
    keywords = [w for w in words if w.lower() not in LEGAL_STOP_WORDS and not w.isdigit()]
    if not keywords:
        return ""
    return " OR ".join(f'"{kw}"' for kw in keywords[:8])


def query_knowledge_graph(facts: QueryFacts, raw_question: str) -> dict[str, Any]:
    """Truy vấn đồ thị Neo4j dựa trên số hiệu Điều (nếu có) hoặc tìm kiếm toàn văn hành vi."""
    # Nếu câu hỏi chứa văn bản không tồn tại -> dừng tra cứu đồ thị để tránh gán ghép sai
    if facts.invalid_docs:
        return {
            "matched_actions": [],
            "penalty_calculations": [],
            "rule_steps": [{
                "rule": "INVALID_DOCUMENT_GUARDRAIL",
                "basis": "Hệ thống văn bản QPPL Việt Nam",
                "detail": f"Văn bản '{', '.join(facts.invalid_docs)}' không tồn tại trong hệ thống pháp luật.",
            }],
            "graph_view": {"nodes": [], "edges": []},
        }

    matched_actions = []
    graph_nodes = []
    graph_edges = []
    seen_node_ids = set()

    def add_node(node_id: str, label: str, node_type: str, data: dict):
        if node_id not in seen_node_ids:
            seen_node_ids.add(node_id)
            graph_nodes.append({
                "id": node_id,
                "label": label,
                "type": node_type,
                "data": data,
            })

    def add_edge(source: str, target: str, rel_type: str):
        graph_edges.append({
            "source": source,
            "target": target,
            "label": rel_type,
        })

    with neo4j_driver.session(database=NEO4J_DATABASE) as session:
        # Trường hợp 1: Người dùng chỉ định rõ số Điều (VD: "Điều 101...")
        if facts.article_refs:
            cypher = """
            MATCH (a:Article) WHERE a.number IN $article_refs
            MATCH (doc:LegalDoc)<-[:PART_OF]-(a)
            MATCH (act:Action)-[:DEFINED_IN]->(k:Clause)-[:PART_OF]->(a)
            OPTIONAL MATCH (act)-[:PENALIZED_BY]->(p:Penalty)
            OPTIONAL MATCH (act)-[:HAS_SANCTION]->(s:Sanction)
            RETURN doc.code AS doc_code, doc.short AS doc_short,
                   a.id AS article_id, a.number AS article_number, a.title AS article_title,
                   k.number AS clause_number, k.text AS clause_text,
                   act.id AS action_id, act.name AS action_name, act.point AS point,
                   p.type AS penalty_type, p.min_amount AS min_amount, p.max_amount AS max_amount,
                   p.warning AS warning, p.base_subject AS base_subject,
                   collect(DISTINCT s.description) AS sanctions
            LIMIT 10
            """
            results = session.run(cypher, article_refs=facts.article_refs).data()
        else:
            # Trường hợp 2: Tìm kiếm theo ngữ nghĩa hành vi bằng Full-text index trên :Action
            ft_query = _sanitize_fulltext_query(raw_question)
            if not ft_query:
                # Không có từ khóa hành vi thực tế -> không trả về hành vi bừa bãi
                results = []
            else:
                cypher = """
                CALL db.index.fulltext.queryNodes('action_fulltext', $query_text)
                YIELD node AS act, score
                MATCH (act)-[:DEFINED_IN]->(k:Clause)-[:PART_OF]->(a:Article)-[:PART_OF]->(doc:LegalDoc)
                OPTIONAL MATCH (act)-[:PENALIZED_BY]->(p:Penalty)
                OPTIONAL MATCH (act)-[:HAS_SANCTION]->(s:Sanction)
                RETURN doc.code AS doc_code, doc.short AS doc_short,
                       a.id AS article_id, a.number AS article_number, a.title AS article_title,
                       k.number AS clause_number, k.text AS clause_text,
                       act.id AS action_id, act.name AS action_name, act.point AS point,
                       p.type AS penalty_type, p.min_amount AS min_amount, p.max_amount AS max_amount,
                       p.warning AS warning, p.base_subject AS base_subject,
                       collect(DISTINCT s.description) AS sanctions,
                       score
                ORDER BY score DESC
                LIMIT 5
                """
                try:
                    results = session.run(cypher, query_text=ft_query).data()
                except Exception as e:
                    print(f"[LỖI Fulltext Neo4j]: {e}")
                    results = []

        # Xây dựng cấu trúc trực quan cho đồ thị và tính toán khung phạt
        penalty_calcs = []
        rule_steps = []

        for row in results:
            doc_code = row["doc_code"]
            art_num = row["article_number"]
            doc_node_id = f"doc:{doc_code}"
            art_node_id = f"art:{row['article_id']}"
            act_node_id = f"act:{row['action_id']}"

            add_node(doc_node_id, row["doc_short"], "LegalDoc", {"code": doc_code})
            add_node(art_node_id, f"Điều {art_num}", "Article", {"title": row["article_title"]})
            add_edge(art_node_id, doc_node_id, "PART_OF")

            action_label = f"K.{row['clause_number']}" + (f".{row['point']}" if row['point'] else "") + f": {row['action_name'][:35]}..."
            add_node(act_node_id, action_label, "Action", {"full_name": row["action_name"]})
            add_edge(act_node_id, art_node_id, "DEFINED_IN")

            penalty_info = {
                "type": row.get("penalty_type") or "FINE",
                "min_amount": row.get("min_amount"),
                "max_amount": row.get("max_amount"),
                "base_subject": row.get("base_subject"),
            }

            # Áp dụng bộ suy luận luật cứng (Deterministic Rule Engine)
            calc = apply_penalty_rules(penalty_info, doc_code, art_num, facts)
            if calc["results"]:
                penalty_calcs.append({
                    "action": row["action_name"],
                    "article": f"Điều {art_num} ({row['doc_short']})",
                    "clause": f"Khoản {row['clause_number']}" + (f" Điểm {row['point']}" if row['point'] else ""),
                    "calculations": calc["results"],
                })
                rule_steps.extend(calc["steps"])

                pen_node_id = f"pen:{row['action_id']}"
                pen_label = "Phạt: " + " | ".join(f"{r['subject_label']}: {r['display']}" for r in calc["results"] if r["applicable"])
                add_node(pen_node_id, pen_label[:60], "Penalty", {"calc": calc["results"]})
                add_edge(act_node_id, pen_node_id, "PENALIZED_BY")

            matched_actions.append({
                "doc_code": doc_code,
                "doc_title": row["doc_short"],
                "article_number": art_num,
                "article_title": row["article_title"],
                "clause_number": row["clause_number"],
                "point": row["point"],
                "action_name": row["action_name"],
                "min_amount": row["min_amount"],
                "max_amount": row["max_amount"],
                "sanctions": [s for s in row["sanctions"] if s],
            })

    # Loại bỏ các bước suy luận trùng lặp
    unique_steps = []
    seen_step_keys = set()
    for s in rule_steps:
        key = (s["rule"], s["detail"])
        if key not in seen_step_keys:
            seen_step_keys.add(key)
            unique_steps.append(s)

    return {
        "matched_actions": matched_actions,
        "penalty_calculations": penalty_calcs,
        "rule_steps": unique_steps,
        "graph_view": {
            "nodes": graph_nodes,
            "edges": graph_edges,
        }
    }

    # GUARDRAIL 2: Phát hiện văn bản ngoài phạm vi chuyên ngành CNTT (ví dụ: Nghị định 100 giao thông)
    if facts.out_of_scope_docs and not facts.article_refs and not _sanitize_fulltext_query(question):
        out_name = ", ".join(facts.out_of_scope_docs)
        guardrail_answer = (
            f"ℹ️ **THÔNG BÁO PHẠM VI TƯ VẤN:**\n\n"
            f"Văn bản **{out_name}** không nằm trong cơ sở dữ liệu chuyên ngành **Công nghệ thông tin & An toàn thông tin mạng**.\n\n"
            f"Hệ thống hiện tại tập trung tư vấn các văn bản nòng cốt:\n"
            f"- **Nghị định 15/2020/NĐ-CP** (Xử phạt vi phạm hành chính lĩnh vực CNTT, viễn thông, mạng xã hội).\n"
            f"- **Luật An ninh mạng 2018 (24/2018/QH14)**.\n"
            f"- **Luật Công nghệ thông tin 2006 (67/2006/QH11)**.\n\n"
            f"👉 Vui lòng đặt câu hỏi liên quan đến lĩnh vực CNTT, an ninh mạng hoặc mạng xã hội."
        )
        return {
            "answer": guardrail_answer,
            "naive_answer": guardrail_answer if request.comparison_mode else None,
            "symbolic_reasoning": {
                "facts": facts.to_dict(),
                "matched_actions": [],
                "penalty_calculations": [],
                "rule_steps": [{
                    "rule": "OUT_OF_SCOPE_GUARDRAIL",
                    "basis": "Phạm vi dữ liệu Đồ án",
                    "detail": f"Văn bản '{out_name}' nằm ngoài phạm vi chuyên môn CNTT & An ninh mạng.",
                }],
            },
            "graph_view": {"nodes": [], "edges": []},
            "sources": [],
        }

    # 2. Luồng Neural: Truy vấn pgvector
    neural_docs = []
    if vector_db:
        try:
            neural_docs = vector_db.similarity_search(question, k=4)
        except Exception as e:
            print(f"[CẢNH BÁO PGVector Search lỗi]: {e}")

    neural_context = "\n\n---\n\n".join(
        [f"[{doc.metadata.get('doc_title', 'Luật')}] {doc.page_content}" for doc in neural_docs]
    ) if neural_docs else "Không tìm thấy đoạn trích văn bản phù hợp trong vector database."

    # 3. Luồng Symbolic: Truy vấn Neo4j & Chạy Rule Engine
    graph_result = query_knowledge_graph(facts, question)

    # Chuẩn bị context văn bản cho Symbolic Reasoning
    symbolic_summary_lines = []
    if graph_result["penalty_calculations"]:
        symbolic_summary_lines.append("CÁC MỨC PHẠT VÀ SUY LUẬN ĐÃ TÍNH TOÁN THEO LUẬT:")
        for calc in graph_result["penalty_calculations"]:
            symbolic_summary_lines.append(f"- Hành vi: {calc['action']}")
            symbolic_summary_lines.append(f"  Căn cứ: {calc['clause']}, {calc['article']}")
            for res in calc["calculations"]:
                symbolic_summary_lines.append(f"  + Chủ thể [{res['subject_label']}]: {res['display']} (Công thức: {res['formula']})")

    if graph_result["rule_steps"]:
        symbolic_summary_lines.append("\nCĂN CỨ NGUYÊN TẮC SUY LUẬN:")
        for step in graph_result["rule_steps"]:
            symbolic_summary_lines.append(f"- Quy tắc [{step['rule']}]: {step['detail']} (Căn cứ: {step['basis']})")

    symbolic_facts_text = "\n".join(symbolic_summary_lines) if symbolic_summary_lines else "Không xác định được điều khoản vi phạm cụ thể trên đồ thị tri thức."

    # 4. Sinh phản hồi Hybrid (Neuro-Symbolic)
    hybrid_prompt_text = prompt_neuro_symbolic.format(
        symbolic_facts=symbolic_facts_text,
        neural_context=neural_context,
        question=question,
    )

    try:
        hybrid_res = llm.invoke(hybrid_prompt_text)
        hybrid_answer = hybrid_res.content if hasattr(hybrid_res, "content") else str(hybrid_res)
    except Exception as e:
        print(f"[LỖI LLM Hybrid]: {e}")
        hybrid_answer = f"Lỗi gọi mô hình ngôn ngữ: {e}"

    # 5. Sinh phản hồi Naive RAG (để so sánh đối chứng A/B nếu bật comparison_mode)
    naive_answer = None
    if request.comparison_mode:
        try:
            naive_prompt_text = prompt_naive.format(
                neural_context=neural_context,
                question=question,
            )
            naive_res = llm.invoke(naive_prompt_text)
            naive_answer = naive_res.content if hasattr(naive_res, "content") else str(naive_res)
        except Exception as e:
            naive_answer = f"(Chế độ Naive RAG gặp sự cố: {e})"

    # Thu thập nguồn trích dẫn
    sources = []
    for doc in neural_docs:
        meta = doc.metadata or {}
        sources.append({
            "title": meta.get("doc_title") or meta.get("doc_code") or "Văn bản luật",
            "article": f"Điều {meta.get('article_number', '')}: {meta.get('article_title', '')}".strip(": "),
            "page": meta.get("page", 1),
            "snippet": doc.page_content[:200] + "...",
        })

    return {
        "answer": hybrid_answer,
        "naive_answer": naive_answer,
        "symbolic_reasoning": {
            "facts": facts.to_dict(),
            "matched_actions": graph_result["matched_actions"],
            "penalty_calculations": graph_result["penalty_calculations"],
            "rule_steps": graph_result["rule_steps"],
        },
        "graph_view": graph_result["graph_view"],
        "sources": sources,
    }

              <Cpu size={18} className="panel-header-icon" />
              <div>
                <h3 className="panel-title">Luồng suy luận Symbolic</h3>
                <span className="panel-sub">Explainable Legal Intelligence (XAI)</span>
              </div>
            </div>
            <button className="panel-close-btn" onClick={() => setShowSymbolicPanel(false)}>
              <X size={16} />
            </button>
          </div>

          {/* Navigation Tabs */}
          <div className="panel-tabs">
            <button 
              className={`panel-tab ${activeTab === 'rules' ? 'active' : ''}`}
              onClick={() => setActiveTab('rules')}
            >
              <GitBranch size={14} />
              <span>Dấu vết Logic</span>
            </button>
            <button 
              className={`panel-tab ${activeTab === 'graph' ? 'active' : ''}`}
              onClick={() => setActiveTab('graph')}
            >
              <Network size={14} />
              <span>Đồ thị Tri thức</span>
            </button>
            <button 
              className={`panel-tab ${activeTab === 'sources' ? 'active' : ''}`}
              onClick={() => setActiveTab('sources')}
            >
              <BookOpen size={14} />
              <span>Trích dẫn Luật</span>
            </button>
          </div>

          <div className="panel-content">
            {!activeSymbolic ? (
              <div className="empty-state">
                <Info size={32} className="empty-icon" />
                <p>Chưa có dữ liệu suy luận cho câu hỏi hiện tại.</p>
                <small>Hãy đặt câu hỏi để quan sát luồng suy luận tự động của hệ thống.</small>
              </div>
            ) : (
              <>
                {/* TAB 1: DẤU VẾT LOGIC & TÍNH TOÁN KHUNG PHẠT */}
                {activeTab === 'rules' && (
                  <div className="rules-view">
                    {/* Các thực thể phát hiện */}
                    <div className="card-box">
                      <div className="card-box-title">
                        <CheckCircle2 size={14} className="icon-success" />
                        <span>Thực thể phát hiện (Query Facts)</span>
                      </div>
                      <div className="facts-grid">
                        <div className="fact-item">
                          <span className="fact-label">Chủ thể nhận diện:</span>
                          <span className="fact-value">
                            {activeSymbolic.reasoning?.facts?.subjects?.map(s => s === 'ORGANIZATION' ? 'Tổ chức' : 'Cá nhân').join(', ') || 'Chung'}
                          </span>
                        </div>
                        {activeSymbolic.reasoning?.facts?.age && (
                          <div className="fact-item">
                            <span className="fact-label">Độ tuổi phát hiện:</span>
                            <span className="fact-value highlight-age">{activeSymbolic.reasoning?.facts?.age} tuổi</span>
                          </div>
                        )}
                        {activeSymbolic.reasoning?.facts?.article_refs?.length > 0 && (
                          <div className="fact-item">
                            <span className="fact-label">Số Điều truy vấn:</span>
                            <span className="fact-value">Điều {activeSymbolic.reasoning?.facts?.article_refs.join(', ')}</span>
                          </div>
                        )}
                      </div>
                    </div>

                    {/* Bảng kết quả tính toán khung phạt theo luật */}
                    {activeSymbolic.reasoning?.penalty_calculations?.length > 0 && (
                      <div className="card-box">
                        <div className="card-box-title">
                          <Scale size={14} className="icon-primary" />
                          <span>Mức phạt tính toán theo quy tắc cứng</span>
                        </div>
                        <div className="calcs-list">
                          {activeSymbolic.reasoning.penalty_calculations.map((calc, i) => (
                            <div key={i} className="calc-card">
                              <div className="calc-action-title">{calc.action}</div>
                              <div className="calc-basis-tag">{calc.clause}, {calc.article}</div>
                              <div className="calc-results-list">
                                {calc.calculations.map((c, j) => (
                                  <div key={j} className="calc-result-row">
                                    <span className="result-subject">{c.subject_label}:</span>
                                    <span className="result-amount">{c.display}</span>
                                    <span className="result-formula">({c.formula})</span>
                                  </div>
                                ))}
                              </div>
                            </div>
                          ))}
                        </div>
                      </div>
                    )}

                    {/* Dấu vết từng bước suy luận (Rule Steps) */}
                    <div className="card-box">
                      <div className="card-box-title">
                        <GitBranch size={14} className="icon-purple" />
                        <span>Quy trình suy luận logic (Rule Trace)</span>
                      </div>
                      <div className="steps-timeline">
                        {activeSymbolic.reasoning?.rule_steps?.map((step, idx) => (
                          <div key={idx} className="timeline-item">
                            <div className="timeline-badge">{idx + 1}</div>
                            <div className="timeline-body">
                              <div className="timeline-rule-tag">[{step.rule}]</div>
                              <div className="timeline-text">{step.detail}</div>
                              <div className="timeline-basis">Căn cứ: <b>{step.basis}</b></div>
                            </div>
                          </div>
                        ))}
                      </div>
                    </div>
                  </div>
                )}

                {/* TAB 2: ĐỒ THỊ TRI THỨC (GRAPH NODES & EDGES) */}
                {activeTab === 'graph' && (
                  <div className="graph-view">
                    <div className="graph-banner">
                      <span>Đã kích hoạt {activeSymbolic.graph?.nodes?.length || 0} Nodes trên Neo4j</span>
                      <a href="http://localhost:7474" target="_blank" rel="noreferrer" className="neo4j-btn">
                        <span>Mở Neo4j Browser</span>
                        <ExternalLink size={12} />
                      </a>
                    </div>

                    <div className="nodes-container">
                      {activeSymbolic.graph?.nodes?.map((node, i) => (
                        <div key={i} className={`graph-node-card type-${node.type?.toLowerCase()}`}>
                          <div className="node-type-badge">{node.type}</div>
                          <div className="node-label-text">{node.label}</div>
                        </div>
                      ))}
                    </div>

                    <div className="edges-section">
                      <h4 className="edges-title">Mối quan hệ liên đới (Relationships):</h4>
                      <div className="edges-list">
                        {activeSymbolic.graph?.edges?.map((edge, i) => (
                          <div key={i} className="edge-item">
                            <span className="edge-node">{edge.source}</span>
                            <span className="edge-rel">--[{edge.label}]--&gt;</span>
                            <span className="edge-node">{edge.target}</span>
                          </div>
                        ))}
                      </div>
                    </div>
                  </div>
                )}

                {/* TAB 3: TRÍCH DẪN VĂN BẢN GỐC (NEURAL SOURCES) */}
                {activeTab === 'sources' && (
                  <div className="sources-view">
                    <div className="sources-list">
                      {activeSymbolic.sources?.map((src, i) => (
                        <div key={i} className="source-card">
                          <div className="source-header">
                            <span className="source-title">{src.title}</span>
                            <span className="source-page">Trang {src.page}</span>
                          </div>
                          <div className="source-article">{src.article}</div>
                          <p className="source-snippet">{src.snippet}</p>
                        </div>
                      ))}
                    </div>
                  </div>
                )}
              </>
            )}
          </div>
        </aside>
      )}
    </div>
  );
}

export default App;
